\begin{definition}
For finite vertex types $V,V'$, a tag type $T$, and maps $s:V\to T$ and
$t:V'\to T$, define the paired-state map $D_{s,t}$ on
$q:T\to\mathbb R\times\mathbb R$ by
\[
D_{s,t}(q)=((\{v:q(s(v))_1=1\},v\mapsto q(s(v))_2),
 (\{v':q(t(v'))_1=1\},v'\mapsto q(t(v'))_2)).
\]
Thus each tag carries a real-coded activation and a priority; the activation
is present precisely when its code equals one.
\end{definition}
