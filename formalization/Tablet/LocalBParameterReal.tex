\begin{definition}
For a finite simple graph $G$, vertex $v$, and real scale $M$, define
\[
b^{\mathbb R}_{G,M}(v)=\frac{\deg_G(v)}M-
\frac{P_G(v)}{M^2}+\frac{T_G(v)}{M^3},
\]
where $P_G(v)$ and $T_G(v)$ count independent pairs and triples in the
neighborhood of $v$. Division uses the convention $x/0=0$; the paper uses
this expression at positive scales bounding the relevant degrees.
\end{definition}
